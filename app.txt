# 사이트별 DB 조회 (HTML용)
# =========================
def get_tabdsp(site_no):
    sql = """
        SELECT
            SITE_NO,
            MATER_NAME,
            PLT_NO,
            QTY,
            MATER_NO,
            BARCODE,
            PROD_DATE,
            VALID_DATE,
            USR_ID,
            CASE
                WHEN CHECK_RESULT = 1 THEN 'OK'
                WHEN CHECK_RESULT = 0 THEN 'NG'
                ELSE 'N/A'
            END AS CHECK_RESULT,
            DNNO,
            SUPPLY_NAME
        FROM TABDSP
        WHERE SITE_NO = :site_no
        ORDER BY PROD_DATE DESC
        
    """

    with pool.acquire() as conn:
        with conn.cursor() as cur:
            cur.execute(sql, site_no=int(site_no))
            row = cur.fetchone()

            if not row:
                return None

            return {
                "site_no": row[0],
                "mater_name": row[1],
                "plt_no": row[2],
                "qty": row[3],
                "mater_no": row[4],
                "barcode": row[5],
                "prod_date": row[6].strftime("%Y-%m-%d") if row[6] else "",
                "valid_date": row[7].strftime("%Y-%m-%d") if row[7] else "",
                "usr_id": row[8],
                "check_result": row[9] or "N/A",
                "dnno": row[10],
                "supply_name": row[11]
            }



CACHE = {}
CACHE_TTL = 5  # 5초

def get_tabdsp_cached(site_no):
    now = time()

    if site_no in CACHE:
        data, ts = CACHE[site_no]
        if now - ts < CACHE_TTL:
            return data

    data = get_tabdsp(site_no)
    CACHE[site_no] = (data, now)
    return data

@app.route("/api/site/<site_id>")
def site_api(site_id):
    data = get_tabdsp(site_id)
    if data:
        return data
    return {"error": "사이트 번호 없음"}

@app.route("/api/display1/<site_id>")
def api_display1(site_id):
    data = get_tabdsp(site_id)
    if data:
        return data
    return {}

@app.route("/api/display2")
def api_display2():
    left_site = request.args.get("left")
    right_site = request.args.get("right")

    result = {}

    if left_site:
        result["left"] = get_tabdsp(left_site)

    if right_site:
        result["right"] = get_tabdsp(right_site)

    return result
